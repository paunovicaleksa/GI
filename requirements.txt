# Nanoplastics scRNA-seq pipeline — direct dependencies
#
# This file lists only what the notebook actually imports, so it resolves on both
# Windows and Linux. It replaces a full `pip freeze` taken on Linux/Python 3.10,
# which is preserved as requirements-lock-linux-py310.txt.
#
# Why not a plain re-freeze: on Windows, jupyter_core pulls in pywin32, which has no
# Linux build at all. A Windows freeze is therefore structurally un-installable on
# Linux, while the direct-dependency list installs cleanly on both.
#
#   python -m venv .venv
#   .\.venv\Scripts\Activate.ps1        # Windows
#   source .venv/bin/activate           # Linux
#   python -m pip install -U pip setuptools wheel
#   python -m pip install -r requirements.txt
#
# Python 3.12 on both platforms. Floor: scanpy and numba both require >=3.10.
# Ceiling: tensorflow 2.17 (via panhumanpy) publishes no wheels above cp312.

# --- load-bearing pins -------------------------------------------------------
# panhumanpy pins tensorflow==2.17 and scikit-learn==1.6.0 exactly. TF 2.17 in turn
# caps numpy below 2.0, which is why numpy sits at 1.26 rather than the 2.2.6 in the
# old Linux freeze. Verified compatible: scanpy needs numpy>=1.24.1 / sklearn>=1.1.3,
# scipy needs numpy>=1.23.5,<2.5, numba needs numpy>=1.22,<2.6 — all lower bounds or
# wide ranges, so nothing else objects to the rollback.
numpy>=1.26,<2.0
scikit-learn==1.6.0
tensorflow==2.17.0

# --- core single-cell stack --------------------------------------------------
scanpy==1.11.5
anndata==0.11.4
h5py==3.16.0
leidenalg==0.12.0          # Leiden clustering (CLAUDE.md prefers it over Louvain)
python-igraph==1.0.0       # leidenalg's graph backend

# Required by sc.pp.scrublet (Section 2), but NOT a hard scanpy dependency. Scrublet's
# automatic threshold detection calls skimage.filters.threshold_minimum, so without
# this the call imports fine and then fails at runtime with
# "threshold is None and thus scrublet requires skimage".
scikit-image

# Required by sc.pp.highly_variable_genes(flavor='seurat_v3') in Section 6, which is
# CLAUDE.md's specified HVG route. Same failure shape as scikit-image above: not a hard
# scanpy dependency, so scanpy imports fine and the call dies at runtime asking for
# skmisc.loess. Ships a cp312 wheel for Windows and Linux; needs numpy>=1.26.4, which
# is exactly the version TensorFlow's ceiling pins us to.
scikit-misc

# --- scientific base ---------------------------------------------------------
scipy==1.15.3
pandas==2.3.3
matplotlib==3.10.9
seaborn==0.13.2

# --- Task 2: batch correction ------------------------------------------------

scanorama

panhumanpy                 # Pan-Human Azimuth; downloads model weights on first use
celltypist                 # immune-specialised second opinion
openpyxl                   # celltypist reads its model metadata from .xlsx

# --- Task 5: differential expression + pathway enrichment --------------------
pydeseq2==0.5.2            # pseudobulk DESeq2 (avoids pseudoreplication)
gseapy==1.3.1              # GSEA primary, ORA secondary

jupyterlab==4.6.3
